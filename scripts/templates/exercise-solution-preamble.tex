% Pandoc may define Highlighting after header includes, but omits it when a
% document has no highlighted code. Define it at begin-document time so the
% environment both exists and has our settings in either case.
\AtBeginDocument{%
  \DefineVerbatimEnvironment{Highlighting}{Verbatim}{%
    commandchars=\\\{\},%
    breaklines=true,%
    breaknonspaceingroup=true,%
    breakanywhere=true,%
    breaksymbolleft={}%
  }%
}

\definecolor{FSMidGray}{HTML}{666666}
\definecolor{FSLightGray}{HTML}{E7E4DF}

\setkomafont{pageheadfoot}{\footnotesize\sffamily\color{FSMidGray}}
\setkomafont{pagefoot}{\footnotesize\sffamily\color{FSMidGray}}

% Keep the exercise hierarchy compact while making each level distinct.
% These settings affect headings in the body; the document title is centered
% separately in before-body.tex.
\setkomafont{section}{\sffamily\bfseries\Large}
\setkomafont{subsection}{\sffamily\bfseries\normalsize}
\setkomafont{subsubsection}{\sffamily\bfseries\small}
\setkomafont{paragraph}{\sffamily\bfseries\small}
\setkomafont{subparagraph}{\sffamily\bfseries\footnotesize}

\KOMAoptions{headsepline=0.3pt}
\setkomafont{headsepline}{\color{FSLightGray}}
